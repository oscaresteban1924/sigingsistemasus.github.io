    codigo: |
      import geopandas as gpd

      # 1. Cargar capas espaciales
      estaciones = gpd.read_file("estaciones.geojson")
      municipios = gpd.read_file("municipios.gpkg")

      print("CRS Estaciones:", estaciones.crs)
      print("CRS Municipios:", municipios.crs)

      # 2. Reproyectar estaciones a EPSG:9377 (MAGNA-SIRGAS Origen Nacional)
      estaciones_proj = estaciones.to_crs(epsg=9377)

      # 3. Joins espaciales (sjoin)
      resultado = gpd.sjoin(estaciones_proj, municipios, how="inner", predicate="within")

      # 4. Exportar a GeoPackage
      resultado.to_file("estaciones_por_municipio.gpkg", driver="GPKG")
      print(f"Procesadas {len(resultado)} estaciones en el territorio.")
    resultadoEsperado: "Nuevo GeoPackage con atributos de estaciones y municipios fusionados en EPSG:9377."
