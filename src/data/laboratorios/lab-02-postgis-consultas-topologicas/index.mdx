      -- Crear tabla espacial de escuelas con SRID 3116 (Metros)
      CREATE TABLE escuelas (
          id SERIAL PRIMARY KEY,
          nombre VARCHAR(100),
          capacidad INT,
          geom GEOMETRY(Point, 3116)
      );

      -- Consulta topológica de proximidad a 200 metros
      SELECT 
          e.nombre AS escuela,
          r.nombre_ruta AS ruta_transporte,
          ST_Distance(e.geom, r.geom) AS distancia_metros
      FROM escuelas e
      JOIN rutas_buses r
        ON ST_DWithin(e.geom, r.geom, 200)
      ORDER BY distancia_metros ASC;
